patterns=[
("basic",r'<!-- BASIC PACKAGE -->(.*?)<!-- STANDARD PACKAGE -->'),

("premium",r'<!-- PREMIUM PACKAGE -->(.*?)<!-- BUSINESS WEBSITE -->'),
("business",r'<!-- BUSINESS WEBSITE -->(.*?)<!-- PROFESSIONAL WEBSITE -->'),
